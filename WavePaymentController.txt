import { Request, Response } from 'express';
import { PrismaClient } from '@prisma/client';
import UCPService from '../services/ucpService';
import WavePaymentService, { WaveCheckoutSessionRequest } from '../services/WavePaymentService';

const prisma = new PrismaClient();

export class WavePaymentController {
  private waveService: WavePaymentService;

  constructor() {
    const baseUrl = process.env.WAVE_API_BASE_URL || 'https://api.wave.com';
    const bearer = process.env.WAVE_CHECKOUT_BEARER;

    const missing: string[] = [];
    if (!bearer) missing.push('WAVE_CHECKOUT_BEARER');
    if (!baseUrl) missing.push('WAVE_API_BASE_URL');
    if (missing.length) {
      throw new Error(`Missing required environment variables: ${missing.join(', ')}`);
    }

    this.waveService = new WavePaymentService({
      baseUrl,
      bearerToken: bearer!,
    });
  }

  /**
   * Create a Wave checkout session for a payment
   * POST /api/payments/wave-gambia/process
   */
  async processPayment(req: Request, res: Response): Promise<void> {
    try {
      const userId = (req as any).user?.id;
      if (!userId) {
        res.status(401).json({ success: false, message: 'User not authenticated' });
        return;
      }

      const {
        amount,
        currency = 'GMD',
        description,
        orderId,
        restrictPayerMobile,
      } = req.body || {};

      if (!amount || Number(amount) <= 0) {
        res.status(400).json({ success: false, message: 'Amount is required and must be > 0' });
        return;
      }

      const originalAmount = Number(amount);
      const currencyCode = String(currency || 'GMD').toUpperCase();

      // Identify order/rental/ride and seller
      let sellerId = userId;
      let actualOrderId: string | null = null;
      let isRental = false;
      let isRide = false;

      if (orderId) {
        const rental = await prisma.rentalRequest.findUnique({
          where: { id: orderId },
          include: { driver: { include: { user: { select: { id: true } } } } },
        });
        if (rental) {
          isRental = true;
          actualOrderId = rental.id;
          sellerId = rental.driver?.user?.id || rental.driver?.userId || sellerId;
          if (rental.customerId !== userId) {
            res.status(403).json({ success: false, message: 'Not authorized to pay for this rental' });
            return;
          }
        } else {
          const ride = await prisma.rideRequest.findUnique({
            where: { requestId: orderId },
            include: { driver: { include: { user: { select: { id: true } } } } },
          });
          if (ride) {
            isRide = true;
            actualOrderId = ride.id;
            sellerId = ride.driver?.user?.id || ride.driver?.userId || sellerId;
            if (ride.customerId !== userId) {
              res.status(403).json({ success: false, message: 'Not authorized to pay for this ride' });
              return;
            }
          } else {
            const order = await prisma.orders.findUnique({
              where: { id: orderId },
              select: { id: true, sellerId: true, userId: true },
            }) || await prisma.orders.findUnique({
              where: { orderNumber: orderId },
              select: { id: true, sellerId: true, userId: true },
            });
            if (!order) {
              res.status(404).json({ success: false, message: 'Order, rental, or ride not found' });
              return;
            }
            actualOrderId = order.id;
            sellerId = order.sellerId || sellerId;
            if (order.userId !== userId) {
              res.status(403).json({ success: false, message: 'Not authorized to pay for this order' });
              return;
            }
          }
        }
      }

      // Generate app-level transaction id
      const appTransactionId = `TXN-${Date.now()}-${Math.random().toString(36).substr(2, 9).toUpperCase()}`;

      // Calculate service fee configuration for Wave (falls back to 0 if not configured)
      const { serviceFeeAmount, serviceFeePercentage, config: serviceFeeConfig } =
        await UCPService.calculateServiceFee('wave_wallet', originalAmount, currencyCode);

      // Build callback/return URLs from a configured public HTTPS app base
      // Never fall back to WAVE_API_BASE_URL (that's the upstream, not our app)
      const rawBase =
        process.env.APP_PUBLIC_BASE_URL ||
        (req.headers.origin as string) ||
        process.env.CLIENT_BASE_URL ||
        process.env.WEB_APP_URL ||
        process.env.FRONTEND_BASE_URL ||
        '';
      let appBase = rawBase ? rawBase.replace(/\/$/, '') : '';
      if (appBase.startsWith('http://')) {
        // Coerce to https to satisfy Wave URL scheme requirement
        appBase = appBase.replace('http://', 'https://');
      }
      if (!appBase || !appBase.startsWith('https://')) {
        res.status(500).json({
          success: false,
          message:
            'APP_PUBLIC_BASE_URL must be configured to a public HTTPS origin for Wave callbacks (e.g. https://your-app.example)',
        });
        return;
      }
      const successUrl = `${appBase}/payments/wave/success`;
      const errorUrl = `${appBase}/payments/wave/error`;

      const payload: WaveCheckoutSessionRequest = {
        amount: String(Math.round(originalAmount)), // Wave examples show integer string
        currency: currencyCode,
        success_url: successUrl,
        error_url: errorUrl,
        client_reference: orderId ? String(orderId) : appTransactionId,
        ...(restrictPayerMobile ? { restrict_payer_mobile: String(restrictPayerMobile) } : {}),
      };

      const session = await this.waveService.createCheckoutSession(payload);

      // Persist external transaction record
      try {
        const transactionData: any = {
          appTransactionId,
          gatewayTransactionId: session.transaction_id || session.id,
          gatewayProvider: 'wave_gambia',
          appService: isRental ? 'RENTAL' : isRide ? 'RIDE' : 'ECOMMERCE',
          amount: originalAmount,
          currencyCode,
          status: 'PENDING',
          paidThroughGateway: true,
          customerId: userId,
          sellerId,
          gatewayResponse: {
            waveSessionId: session.id,
            waveLaunchUrl: session.wave_launch_url,
            checkoutStatus: session.checkout_status,
            paymentStatus: session.payment_status,
            clientReference: session.client_reference,
            successUrl,
            errorUrl,
            serviceFeeAmount,
            serviceFeePercentage,
            serviceFeeConfig: serviceFeeConfig?.name || 'service_fee_wave_wallet',
            description,
          },
        };

        if (isRental) transactionData.rentalRequestId = actualOrderId;
        else if (isRide) transactionData.rideRequestId = actualOrderId;
        else transactionData.orderId = actualOrderId;

        await prisma.externalTransaction.create({ data: transactionData });

        if (serviceFeeAmount > 0) {
          const serviceFeeTransactionId = `SVC_${Date.now()}_${Math.random().toString(36).substr(2, 9).toUpperCase()}`;
          const serviceFeeData: any = {
            appTransactionId: serviceFeeTransactionId,
            gatewayTransactionId: session.transaction_id || session.id,
            gatewayProvider: 'wave_gambia',
            appService: isRental ? 'RENTAL' : isRide ? 'RIDE' : 'ECOMMERCE',
            amount: serviceFeeAmount,
            currencyCode,
            status: 'PENDING',
            transactionType: 'SERVICE_FEE',
            customerId: userId,
            sellerId,
            gatewayResponse: {
              parentWaveSessionId: session.id,
              serviceFeePercentage,
              serviceFeeConfig: serviceFeeConfig?.name || 'service_fee_wave_wallet',
              description: `Service fee for Wave payment ${session.id}`,
            },
          };
          if (isRental) serviceFeeData.rentalRequestId = actualOrderId;
          else if (isRide) serviceFeeData.rideRequestId = actualOrderId;
          else serviceFeeData.orderId = actualOrderId;
          await prisma.externalTransaction.create({ data: serviceFeeData });
        }
      } catch (dbErr) {
        console.error('Error creating Wave external transaction record:', dbErr);
      }

      res.status(200).json({
        success: true,
        data: {
          sessionId: session.id,
          transactionId: session.transaction_id,
          status: session.payment_status,
          checkoutStatus: session.checkout_status,
          waveLaunchUrl: session.wave_launch_url,
          amount: Number(session.amount),
          currency: session.currency,
          serviceFee: {
            amount: serviceFeeAmount,
            percentage: serviceFeePercentage,
            config: serviceFeeConfig?.name || 'service_fee_wave_wallet',
          },
          message: 'Wave checkout session created',
        },
      });
    } catch (error: any) {
      console.error('Wave processPayment error:', error?.response?.data || error?.message || error);
      res.status(500).json({
        success: false,
        message: 'Failed to create Wave checkout session',
        error: error?.response?.data || error?.message,
      });
    }
  }

  /**
   * GET /api/payments/wave-gambia/sessions/:id
   */
  async getSession(req: Request, res: Response): Promise<void> {
    try {
      const { id } = req.params;
      if (!id) {
        res.status(400).json({ success: false, message: 'Session ID is required' });
        return;
      }
      const session = await this.waveService.getCheckoutSession(id);
      res.status(200).json({ success: true, data: session });
    } catch (error: any) {
      console.error('Wave getSession error:', error?.response?.data || error?.message || error);
      res.status(500).json({
        success: false,
        message: 'Failed to fetch session',
        error: error?.response?.data || error?.message,
      });
    }
  }

  /**
   * POST /api/payments/wave-gambia/sessions/:id/expire
   */
  async expireSession(req: Request, res: Response): Promise<void> {
    try {
      const { id } = req.params;
      if (!id) {
        res.status(400).json({ success: false, message: 'Session ID is required' });
        return;
      }
      const session = await this.waveService.expireCheckoutSession(id);
      res.status(200).json({ success: true, data: session });
    } catch (error: any) {
      console.error('Wave expireSession error:', error?.response?.data || error?.message || error);
      res.status(500).json({
        success: false,
        message: 'Failed to expire session',
        error: error?.response?.data || error?.message,
      });
    }
  }

  /**
   * POST /api/payments/wave-gambia/sessions/:id/refund
   */
  async refundSession(req: Request, res: Response): Promise<void> {
    try {
      const { id } = req.params;
      if (!id) {
        res.status(400).json({ success: false, message: 'Session ID is required' });
        return;
      }
      const session = await this.waveService.refundCheckoutSession(id);
      res.status(200).json({ success: true, data: session });
    } catch (error: any) {
      console.error('Wave refundSession error:', error?.response?.data || error?.message || error);
      res.status(500).json({
        success: false,
        message: 'Failed to refund session',
        error: error?.response?.data || error?.message,
      });
    }
  }

  /**
   * GET /api/payments/wave-gambia/currencies
   */
  async getSupportedCurrencies(_req: Request, res: Response): Promise<void> {
    try {
      res.status(200).json({
        success: true,
        data: {
          currencies: [{ code: 'GMD', name: 'Gambian Dalasi', symbol: 'D' }],
          default: 'GMD',
        },
      });
    } catch (error: any) {
      res.status(500).json({ success: false, message: 'Internal server error', error: error.message });
    }
  }

  /**
   * GET /api/payments/wave-gambia/check-transactions/:orderId
   */
  async checkExistingTransactions(req: Request, res: Response): Promise<void> {
    try {
      const { orderId } = req.params;
      if (!orderId) {
        res.status(400).json({ success: false, message: 'Order ID is required' });
        return;
      }

      const existingTransactions = await prisma.externalTransaction.findMany({
        where: {
          orderId: orderId,
          gatewayProvider: 'wave_gambia',
          status: { notIn: ['FAILED', 'CANCELLED'] },
        },
        select: {
          id: true,
          status: true,
          gatewayProvider: true,
          amount: true,
          currencyCode: true,
          createdAt: true,
          transactionType: true,
        },
        orderBy: { createdAt: 'desc' },
      });

      const hasActiveTransaction = existingTransactions.some(
        (tx) => tx.status === 'PENDING' || tx.status === 'SUCCESS'
      );

      res.status(200).json({
        success: true,
        data: {
          hasActiveTransaction,
          transactions: existingTransactions,
          canMakePayment: !hasActiveTransaction,
        },
      });
    } catch (error: any) {
      console.error('Wave checkExistingTransactions error:', error);
      res.status(500).json({
        success: false,
        message: 'Failed to check existing transactions',
        error: error.message,
      });
    }
  }
}

export default WavePaymentController;
